# AI 应用生产化实战指南

## 概述

本文是阶段六的综合实战指南，目标是将前面阶段的 AI 项目进行生产化改造：

```
Mini ChatGPT  →  Docker 化 + 日志 + 重试 + 缓存 + 费用追踪
Mini Dify     →  Docker Compose + Redis Queue
RAG 系统      →  Guardrails + Token 优化 + 链路追踪
```

---

## 一、项目改造计划

### 1.1 改造清单

每个项目需要完成以下改造：

```
[ ] 编写 Dockerfile
[ ] 加入结构化日志
[ ] 加入 Retry + Fallback
[ ] 加入 Redis 异步队列（耗时任务）
[ ] 加入 Cost Monitor（费用追踪）
[ ] 编写 CI 工作流
[ ] 编写 docker-compose.yml
```

### 1.2 改造顺序

```
第 1 步：日志 + 重试（最基础，改动最小）
第 2 步：Cost Monitor（了解花费）
第 3 步：Docker 化（标准化部署）
第 4 步：Redis Queue（异步任务）
第 5 步：Docker Compose（多服务编排）
第 6 步：CI/CD（自动化）
```

### 2.2 LLM 调用模块（带重试 + 费用追踪）

```javascript
// server/llm/client.js
const Anthropic = require('@anthropic-ai/sdk');
const { logger } = require('../middleware/logger');
const { costTracker } = require('./cost');
const { cache } = require('./cache');

const FALLBACK_MODELS = [
  'claude-sonnet-4-6',
  'claude-haiku-4-5-20251001',
];

async function resilientCall(messages, options = {}) {
  const { requestId = 'unknown', maxRetries = 2, useCache = true } = options;

  // 1. 查缓存
  if (useCache) {
    const cached = cache.get(messages);
    if (cached) {
      logger.info('cache.hit', { requestId });
      return cached;
    }
  }

  // 2. Fallback 链
  let lastError;
  for (const model of FALLBACK_MODELS) {
    // 3. Retry 循环
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        logger.info('llm.request', { requestId, model, attempt });

        const client = new Anthropic();
        const start = Date.now();

        const response = await client.messages.create({
          model,
          max_tokens: options.maxTokens || 2048,
          messages,
        });

        const duration = Date.now() - start;

        logger.info('llm.response', {
          requestId,
          model,
          durationMs: duration,
          inputTokens: response.usage.input_tokens,
          outputTokens: response.usage.output_tokens,
        });

        // 4. 记录费用
        costTracker.record({
          model,
          inputTokens: response.usage.input_tokens,
          outputTokens: response.usage.output_tokens,
          requestId,
        });

        // 5. 写缓存
        if (useCache) {
          cache.set(messages, response);
        }

        return response;

      } catch (error) {
        lastError = error;
        const duration = Date.now() - start;

        if (!isRetryable(error) || attempt === maxRetries) {
          logger.error('llm.error', {
            requestId, model, attempt,
            durationMs: duration,
            error: error.message,
          });
          break; // 切换下一个模型
        }

        const delay = 1000 * Math.pow(2, attempt);
        logger.warning('llm.retry', { requestId, model, attempt, delayMs: delay });
        await sleep(delay);
      }
    }
  }

  throw lastError;
}

function isRetryable(error) {
  const status = error.status || 0;
  return status === 429 || (status >= 500 && status < 600);
}

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

module.exports = { resilientCall };
```

### 2.3 费用追踪模块

```javascript
// server/llm/cost.js
const PRICING = {
  'claude-opus-4-7': { input: 15, output: 75 },
  'claude-sonnet-4-6': { input: 3, output: 15 },
  'claude-haiku-4-5-20251001': { input: 0.25, output: 1.25 },
};

class CostTracker {
  constructor() {
    this.records = [];
  }

  record({ model, inputTokens, outputTokens, requestId }) {
    const pricing = PRICING[model] || { input: 3, output: 15 };
    const cost = (inputTokens * pricing.input + outputTokens * pricing.output) / 1_000_000;

    this.records.push({
      timestamp: new Date().toISOString(),
      model,
      inputTokens,
      outputTokens,
      costUsd: cost,
      requestId,
    });

    return cost;
  }

  getDailySummary() {
    const today = new Date().toISOString().split('T')[0];
    const todayRecords = this.records.filter(
      r => r.timestamp.startsWith(today)
    );

    const totalCost = todayRecords.reduce((sum, r) => sum + r.costUsd, 0);
    const totalCalls = todayRecords.length;

    const byModel = {};
    for (const r of todayRecords) {
      if (!byModel[r.model]) byModel[r.model] = { cost: 0, calls: 0 };
      byModel[r.model].cost += r.costUsd;
      byModel[r.model].calls += 1;
    }

    return {
      date: today,
      totalCostUsd: Math.round(totalCost * 10000) / 10000,
      totalCalls,
      byModel,
    };
  }
}

const costTracker = new CostTracker();
module.exports = { costTracker };
```

### 2.4 日志中间件

```javascript
// server/middleware/logger.js
const winston = require('winston');
const crypto = require('crypto');

const logger = winston.createLogger({
  level: 'info',
  format: winston.format.combine(
    winston.format.timestamp(),
    winston.format.json(),
  ),
  transports: [
    new winston.transports.Console(),
    new winston.transports.File({ filename: 'logs/app.log' }),
  ],
});

function requestLogger(req, res, next) {
  const requestId = crypto.randomUUID().slice(0, 12);
  req.requestId = requestId;
  req.startTime = Date.now();

  logger.info('request.start', {
    requestId,
    method: req.method,
    path: req.path,
    contentLength: req.headers['content-length'],
  });

  const originalEnd = res.end;
  res.end = function (...args) {
    const duration = Date.now() - req.startTime;
    logger.info('request.end', {
      requestId,
      statusCode: res.statusCode,
      durationMs: duration,
    });
    originalEnd.apply(res, args);
  };

  next();
}

module.exports = { logger, requestLogger };
```

### 3.2 RAG 请求的完整管道

```python
import time
import logging

log = logging.getLogger("rag")

async def rag_pipeline(question, vector_store):
    """生产级 RAG 管道"""
    trace_id = generate_trace_id()
    start = time.time()

    # 1. 输入 Guardrails
    guard_result = check_input(question)
    if not guard_result.passed:
        log.warning("guardrail.blocked", trace_id=trace_id, reason=guard_result.message)
        return {"error": guard_result.message}

    # 2. Token 预算检查
    budget = cost_tracker.check_budget()
    if budget["over_budget"]:
        log.warning("cost.over_budget", trace_id=trace_id, **budget)
        return {"error": "服务暂时不可用，请稍后重试"}

    # 3. 向量检索
    t_search = time.time()
    docs = vector_store.search(question, top_k=5)
    search_ms = (time.time() - t_search) * 1000

    log.info("search.done", trace_id=trace_id, results=len(docs), duration_ms=round(search_ms))

    # 4. Prompt 组装（压缩上下文）
    context = "\n".join([d["content"][:500] for d in docs])  # 截断单个文档
    prompt = f"基于以下资料回答：\n{context}\n\n问题：{question}"

    # 5. LLM 调用（带重试 + Fallback + 费用追踪）
    response = await resilient_call(
        [{"role": "user", "content": prompt}],
        request_id=trace_id,
    )

    output_text = response.content[0].text

    # 6. 输出 Guardrails
    safety_result = check_output(output_text)
    if not safety_result.passed:
        log.warning("guardrail.output_blocked", trace_id=trace_id, reason=safety_result.message)
        return {"error": "无法生成回答，请换个问题"}

    total_ms = (time.time() - start) * 1000

    log.info("rag.complete", trace_id=trace_id, total_ms=round(total_ms),
             search_ms=round(search_ms), sources=len(docs))

    return {
        "answer": output_text,
        "sources": [{"content": d["content"][:200], "score": d.get("score")} for d in docs],
        "trace_id": trace_id,
    }
```

### 4.1 仓库级 CI

```yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

jobs:
  # Python 测试
  test-python:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip
      - run: pip install -r requirements.txt
      - run: pytest tests/ -v

  # Node.js 测试
  test-node:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
      - run: |
          for dir in code/chatgpt code/minidify code/mini-cursor; do
            if [ -f "$dir/package.json" ]; then
              cd "$dir"
              npm ci
              npm run lint
              npm run build
              cd -
            fi
          done

  # 构建 Docker 镜像
  build:
    runs-on: ubuntu-latest
    needs: [test-python, test-node]
    if: github.ref == 'refs/heads/main'
    steps:
      - uses: actions/checkout@v4
      - uses: docker/setup-buildx-action@v3
      - uses: docker/login-action@v3
        with:
          username: ${{ secrets.DOCKER_USERNAME }}
          password: ${{ secrets.DOCKER_PASSWORD }}
      - uses: docker/build-push-action@v5
        with:
          context: ./code/chatgpt
          push: true
          tags: myorg/ai-chatgpt:latest
          cache-from: type=gha
          cache-to: type=gha,mode=max

  # 部署
  deploy:
    runs-on: ubuntu-latest
    needs: [build]
    if: github.ref == 'refs/heads/main'
    steps:
      - uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.SERVER_HOST }}
          username: ${{ secrets.SERVER_USER }}
          key: ${{ secrets.SSH_PRIVATE_KEY }}
          script: |
            cd /opt/ai-app
            docker compose pull
            docker compose up -d
```

## 小结

生产化不是一步到位的，而是逐步叠加：

```
第一层：重试 + 日志（1-2 小时）
第二层：费用追踪 + Guardrails（2-3 小时）
第三层：Docker 化（2-3 小时）
第四层：Redis Queue（3-4 小时）
第五层：CI/CD（2-3 小时）
```

每一层独立可测试，不会互相阻塞。建议按顺序叠加，每完成一层就验证一次。
