## 前置回顾

上一节我们学了 DAG 和状态机。但在实际 Workflow 中，还有两个关键问题：

```
1. 节点怎么调度？（先执行谁？后执行谁？可以并行吗？）
2. 怎么根据条件选择路径？（如果A就左转，如果B就右转）
```

这就是**节点调度**和**条件流转**。

## 节点调度

### 什么是节点调度

> 决定 Workflow 中各个节点**什么时候执行、以什么顺序执行**。

### 调度策略

#### 1. 顺序执行（Sequential）

最简单的方式——一个接一个执行：

```python
def sequential_execute(nodes, input_data):
    """顺序执行所有节点"""
    result = input_data

    for node in nodes:
        result = node.execute(result)

    return result

# 示例
nodes = [load_doc, extract_text, summarize, format_output]
result = sequential_execute(nodes, {"file": "report.pdf"})
```

```
输入 → [节点A] → [节点B] → [节点C] → 输出
```

#### 2. 并行执行（Parallel）

互不依赖的节点同时执行：

```python
import concurrent.futures

def parallel_execute(dag, input_data):
    """并行执行无依赖的节点"""
    completed = {}
    pending = set(dag.nodes.keys())

    while pending:
        # 找出所有依赖已完成的节点
        ready = [
            name for name in pending
            if all(parent.name in completed for parent in dag.nodes[name].parents)
        ]

        # 并行执行这些节点
        with concurrent.futures.ThreadPoolExecutor() as executor:
            futures = {}
            for name in ready:
                node = dag.nodes[name]
                inputs = {p.name: completed[p.name] for p in node.parents}
                if not inputs:
                    inputs = input_data
                futures[executor.submit(node.handler, inputs)] = name

            for future in concurrent.futures.as_completed(futures):
                name = futures[future]
                completed[name] = future.result()
                pending.remove(name)

    return completed
```

```
           ┌→ [节点B] →┐
输入 → [节点A]            → [节点D] → 输出
           └→ [节点C] →┘
         (B和C并行执行)
```

#### 3. 拓扑排序

DAG 的经典调度算法——拓扑排序：

```python
def topological_sort(dag):
    """拓扑排序：确定节点的执行顺序"""
    in_degree = {name: len(node.parents) for name, node in dag.nodes.items()}
    queue = [name for name, degree in in_degree.items() if degree == 0]
    order = []

    while queue:
        # 取出一个入度为0的节点
        node_name = queue.pop(0)
        order.append(node_name)

        # 减少下游节点的入度
        for child in dag.nodes[node_name].children:
            in_degree[child.name] -= 1
            if in_degree[child.name] == 0:
                queue.append(child.name)

    return order

# 示例
order = topological_sort(dag)
# → ['load_doc', 'extract_text', 'translate', 'summarize', 'output']
# 按这个顺序执行，保证每个节点的依赖已完成
```

## 条件流转

### 什么是条件流转

> 根据上一步的结果，**动态决定**下一步走哪条路径。

这是 Workflow 和简单 DAG 的关键区别——**if/else 逻辑**。

### 条件流转的表示

```
              ┌ 是 → [处理退款]
[判断金额] ──┤
              └ 否 → [转人工客服]
```

### 代码实现

```python
class ConditionNode:
    """条件节点：根据条件选择路径"""
    def __init__(self, name, condition_fn):
        self.name = name
        self.condition_fn = condition_fn  # 条件判断函数
        self.branches = {}                # 条件结果 → 下一个节点

    def add_branch(self, condition_result, next_node):
        self.branches[condition_result] = next_node

    def execute(self, input_data):
        result = self.condition_fn(input_data)
        next_node = self.branches.get(result)

        if next_node:
            return next_node.execute(input_data)
        return input_data


class ActionNode:
    """动作节点：执行一个具体的操作"""
    def __init__(self, name, handler):
        self.name = name
        self.handler = handler

    def execute(self, input_data):
        return self.handler(input_data)


# 使用示例：构建一个智能客服流程
def check_intent(input_data):
    """判断用户意图"""
    return input_data.get("intent")  # "refund", "complaint", "query"

def check_amount(input_data):
    """判断金额是否超过阈值"""
    amount = input_data.get("amount", 0)
    return "auto" if amount < 100 else "manual"

# 构建流程
intent_check = ConditionNode("intent_check", check_intent)
amount_check = ConditionNode("amount_check", check_amount)

refund_auto = ActionNode("refund_auto", lambda d: {"response": "已自动退款"})
refund_manual = ActionNode("refund_manual", lambda d: {"response": "已转人工处理"})
complaint_handler = ActionNode("complaint", lambda d: {"response": "已记录投诉"})
query_handler = ActionNode("query", lambda d: {"response": "查询结果..."})

# 连接条件分支
intent_check.add_branch("refund", amount_check)
intent_check.add_branch("complaint", complaint_handler)
intent_check.add_branch("query", query_handler)

amount_check.add_branch("auto", refund_auto)
amount_check.add_branch("manual", refund_manual)
```

流程图：
```
                    ┌─ "refund" → [判断金额]
                    │                ├─ auto → [自动退款]
[判断意图] ─────────┤                └─ manual → [人工处理]
                    ├─ "complaint" → [处理投诉]
                    └─ "query" → [回答问题]
```

## 完整的 Workflow 引擎

把调度和条件流转组合起来：

```python
class WorkflowEngine:
    """简易 Workflow 执行引擎"""

    def __init__(self):
        self.nodes = {}
        self.edges = {}          # node_name → [{condition, target}]
        self.start_node = None

    def add_node(self, name, handler, node_type="action"):
        """添加节点"""
        self.nodes[name] = {
            "name": name,
            "type": node_type,       # "action" 或 "condition"
            "handler": handler,
        }
        self.edges[name] = []

    def add_edge(self, from_node, to_node, condition=None):
        """添加边，可带条件"""
        self.edges[from_node].append({
            "condition": condition,
            "target": to_node
        })

    def set_start(self, node_name):
        self.start_node = node_name

    def run(self, input_data):
        """执行工作流"""
        current = self.start_node
        context = dict(input_data)

        while current:
            node = self.nodes[current]
            result = node["handler"](context)

            # 更新上下文
            context.update(result if isinstance(result, dict) else {"result": result})

            # 找下一个节点
            edges = self.edges[current]
            next_node = None

            for edge in edges:
                if edge["condition"] is None:
                    # 无条件 → 直接连线
                    next_node = edge["target"]
                elif edge["condition"](context):
                    # 条件满足 → 走这条边
                    next_node = edge["target"]
                    break

            current = next_node

        return context


# 使用示例
engine = WorkflowEngine()

engine.add_node("start", lambda ctx: {"step": "started"})
engine.add_node("analyze", lambda ctx: {"intent": "refund", "amount": 50})
engine.add_node("auto_refund", lambda ctx: {"response": "已退款", "done": True})
engine.add_node("manual_review", lambda ctx: {"response": "转人工", "done": True})
engine.add_node("end", lambda ctx: ctx)

engine.add_edge("start", "analyze")
engine.add_edge("analyze", "auto_refund",
                condition=lambda ctx: ctx.get("amount", 0) < 100)
engine.add_edge("analyze", "manual_review",
                condition=lambda ctx: ctx.get("amount", 0) >= 100)
engine.add_edge("auto_refund", "end")
engine.add_edge("manual_review", "end")

engine.set_start("start")

result = engine.run({"user_input": "我要退款"})
# → {"user_input": "我要退款", "intent": "refund", "amount": 50, "response": "已退款", "done": True}
```

## 常见工作流模式

### 1. 串行模式（Sequential）
```
[A] → [B] → [C] → [D]
```

### 2. 并行模式（Parallel）
```
[A] → [B]
[A] → [C]
[B] → [D]
[C] → [D]
```

### 3. 条件分支（Conditional）
```
[A] → 条件判断 → [B] 或 [C]
```

### 4. 循环模式（Loop）
```
[A] → [B] → 条件 → [C]（完成）
                  → [B]（重试）
```

### 5. 子流程（Sub-Workflow）
```
[A] → [子流程: X→Y→Z] → [B]
```
